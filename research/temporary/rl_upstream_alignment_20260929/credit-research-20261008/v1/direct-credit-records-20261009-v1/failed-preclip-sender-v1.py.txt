"""CPU-test the passive norm seam, then submit one native-worker install RPC."""
import hashlib
import importlib.util
import json
from pathlib import Path
import subprocess

HERE=Path(__file__).resolve().parent
spec=importlib.util.spec_from_file_location('transport',HERE.parents[1]/'stage_environment_entry.py')
transport=importlib.util.module_from_spec(spec);spec.loader.exec_module(transport)
out=transport.ROOT+'/receipts/textcraft-native-actor-incidents-20261010-v1/preclip-v1'
receipt=HERE/'direct-credit-records-20261009-v1/native-preclip-stage-20261010.json'
assert not receipt.exists(), 'Inspect the existing submitted handle instead of resubmitting'

def remote(code):
    command=('source '+transport.ENTRY+'/metax-entry.env.sh\n'
             'CUDA_VISIBLE_DEVICES=-1 "$VENV_PYTHON" - <<\'PY\'\n'+code+'\nPY\n')
    result=subprocess.run(transport.SSH+['bash','-s'],input=command.encode(),capture_output=True,timeout=55)
    result.check_returncode()
    return json.loads(result.stdout)

remote('from pathlib import Path\nimport json\np=Path('+repr(out)+')\np.mkdir(parents=True,exist_ok=True)\nprint(json.dumps(dict(path=str(p))))')
files={}
for name in ['capture_native_preclip_20261010.py','test_capture_native_preclip_20261010.py']:
    path=HERE/name
    subprocess.run(transport.SCP+[str(path),transport.SSH[-1]+':'+out+'/'+name],check=True,timeout=30)
    files[name]=hashlib.sha256(path.read_bytes()).hexdigest()
check=remote('import hashlib,json,os,subprocess,sys\nfrom pathlib import Path\np=Path('+repr(out)+')\n'
 'expected='+repr(files)+'\n'
 'for name,sha in expected.items():assert hashlib.sha256((p/name).read_bytes()).hexdigest()==sha\n'
 'r=subprocess.run([sys.executable,str(p/"test_capture_native_preclip_20261010.py")],capture_output=True,text=True,timeout=35)\n'
 'r.check_returncode()\nprint(r.stdout)')
receipt.write_text(json.dumps(dict(status='CPU_interface_test_passed_not_installed',files=files,cpu_check=check),indent=2)+'\n')

launch=remote('import hashlib,json,psutil,subprocess,sys,time\nfrom pathlib import Path\n'
 'out=Path('+repr(out)+')\n'
 'driver=psutil.Process(982372);assert driver.create_time()==1791553809.84\n'

data=dict(status='submitted_native_RPC_not_verified',files=files,cpu_check=check,launch=launch,
